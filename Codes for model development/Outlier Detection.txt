def set_ylim_safe(ax, values, default=(0.0, 1.0), buffer_ratio=0.1, min_padding=0.02):
    vals = [float(v) for v in values if np.isfinite(v)]
    if not vals:
        ax.set_ylim(*default)
        return
    lo, hi = min(vals), max(vals)
    if lo == hi:
        pad = max(min_padding, abs(lo) * 0.05)
        ax.set_ylim(lo - pad, hi + pad)
    else:
        pad = (hi - lo) * buffer_ratio
        ax.set_ylim(lo - pad, hi + pad)

def load_data_safely(file_path: str, delimiter: str, expected_cols: int) -> np.ndarray:
    try:
        df = pd.read_csv(
            file_path,
            delimiter=delimiter,
            header=None,
            skip_blank_lines=True,
            error_bad_lines=False,
            warn_bad_lines=True,
            encoding=ENCODING,
            dtype=str
        )
        df = df.replace('#NUM!', np.nan).replace('', np.nan)
        df = df.applymap(lambda x: np.nan if isinstance(x, str) and x.strip() == '' else x)
        df = df.astype(np.float64)
        row_col_counts = df.notna().sum(axis=1)
        valid_rows = row_col_counts == expected_cols
        df = df[valid_rows].reset_index(drop=True)
        data = df.values
        if data.ndim != 2 or data.shape[1] != expected_cols:
            raise ValueError(f"数据形状错误 {data.shape}")
        return data
    except Exception as e:
        raise RuntimeError(f"数据读取失败：{str(e)}") from e

def run_lof(k: int, data: np.ndarray, contamination: float) -> Dict[str, Union[int, float, np.ndarray]]:
    lof = LocalOutlierFactor(
        n_neighbors=k,
        contamination=contamination,
        novelty=False,
        n_jobs=-1
    )
    y_pred = lof.fit_predict(data)
    scores = -lof.negative_outlier_factor_
    out_idx = np.where(y_pred == -1)[0]
    n_out = len(out_idx)
    out_ratio = n_out / len(data)
    score_threshold = np.min(scores[out_idx]) if n_out > 0 else np.nan
    return {
        "k": k,
        "n_outliers": n_out,
        "outlier_ratio": out_ratio,
        "score_mean_outliers": scores[out_idx].mean() if n_out > 0 else np.nan,
        "score_mean_inliers": scores[np.where(y_pred == 1)[0]].mean() if (len(data) - n_out) > 0 else np.nan,
        "score_threshold": score_threshold,
        "outlier_indices": out_idx,
        "all_scores": scores
    }

results: List[Dict] = []
for k in NEIGHBORS_LIST:
    try:
        res = run_lof(k, data, CONTAMINATION)
        results.append(res)
        print(f"[k={k:>2}] 异常数={res['n_outliers']:>3}, 占比={res['outlier_ratio']:.2%}")
    except Exception as e:
        print(f"[ERROR] k={k} 执行失败：{str(e)}")

if not results:

def jaccard_similarity(set_a: np.ndarray, set_b: np.ndarray) -> float:
    a, b = set(set_a.tolist()), set(set_b.tolist())
    return len(a & b) / len(a | b) if (a or b) else 1.0

def score_correlation(score_a: np.ndarray, score_b: np.ndarray) -> float:
    return np.corrcoef(score_a, score_b)[0, 1]

stability_metrics = {"jaccard_vs_prev": [np.nan], "score_corr_vs_prev": [np.nan]}
for i in range(1, len(results)):

    stability_metrics["jaccard_vs_prev"].append(
        jaccard_similarity(prev_res["outlier_indices"], curr_res["outlier_indices"]))
    stability_metrics["score_corr_vs_prev"].append(
        score_correlation(prev_res["all_scores"], curr_res["all_scores"]))

def select_suggest_k(results: List[Dict], stability_metrics: Dict, min_jaccard: float = 0.6) -> int:
    temp_data = []
    for i, res in enumerate(results):
        temp_data.append({
            "k": res["k"],
            "out_ratio": res["outlier_ratio"],
            "jaccard": stability_metrics["jaccard_vs_prev"][i],
            "ratio_deviation": abs(res["outlier_ratio"] - CONTAMINATION)
        })
    temp_df = pd.DataFrame(temp_data)
    candidate_mask = temp_df["jaccard"] >= min_jaccard
    if not candidate_mask.any():
        candidate_mask[:] = True
    candidates = temp_df[candidate_mask].sort_values(by=["ratio_deviation", "k"])
    return int(candidates.iloc[0]["k"])
