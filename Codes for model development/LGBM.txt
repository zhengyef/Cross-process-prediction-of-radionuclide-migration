def plot_fig(num, model, n, data1, data2, data3, dpi=500):
    explainer = shap.TreeExplainer(model)
    if (num - 1) == n:
        shap_values = explainer.shap_values(data1)
        shap.initjs()
        Y_pred_test = model.predict(data2)
        score_test = r2_score(data3, Y_pred_test)
        print("第" + str(round(n+1)) + "次的预测R2:", score_test)

        n_features = data1.shape[1]  

        fig = plt.figure()
        shap.summary_plot(shap_values, data1, plot_type='bar', max_display=n_features)
        fig = plt.figure()
        shap.summary_plot(shap_values[:, :], data1.iloc[:, :], max_display=n_features)


        feature_importance = model.feature_importance(importance_type='gain')  # 或者使用 'split' 或 'weight'
        sorted_indices = np.argsort(feature_importance)
        sorted_feature_importance = feature_importance[sorted_indices]
        sorted_feature_names = [data.columns[i] for i in sorted_indices]


        plt.figure(figsize=(20, 16))
        plt.barh(range(len(sorted_feature_names)), sorted_feature_importance, color='b', align='center')
        plt.yticks(range(len(sorted_feature_names)), sorted_feature_names)
        plt.xlabel('Feature Importance')
        plt.ylabel('Feature Name')
        plt.title('Feature Importance')
        plt.show()

 
        shap.plots.partial_dependence('Dw', model.predict, data1, ice=False)
        shap.plots.partial_dependence('redox', model.predict, data1, ice=False)
        

        print("绘制完毕！")

       
        explainer = shap.TreeExplainer(model)
        shap_values_numpy = explainer.shap_values(X)

        fig, ax1 = plt.subplots(figsize=(25, 15), dpi=1200)

        plt.sca(ax1)
        shap.summary_plot(shap_values_numpy, X, feature_names=X.columns,
                          plot_type="dot", show=False, color_bar=True, max_display=n_features)
        ax1.set_position([0.1, 0.1, 0.65, 0.8])  

 
        ax2 = ax1.twiny()  
        plt.sca(ax2)
        shap.summary_plot(shap_values_numpy, X, plot_type="bar",
                          show=False, max_display=n_features)


        ax2.set_position(ax1.get_position()) 
        ax2.xaxis.set_label_position('top')  
        ax2.xaxis.tick_top() 


        ax2.axhline(y=13, color='gray', linestyle='-', linewidth=1)


        bars = ax2.patches
        for bar in bars:
            bar.set_alpha(0.2)


        ax1.set_xlabel('Shapley Value Contribution (Bee Swarm)', fontsize=12)
        ax2.set_xlabel('Mean Shapley Value (Feature Importance)', fontsize=12)
        ax1.set_ylabel('Features', fontsize=12)


        plt.tight_layout()
        plt.savefig("SHAP_combined_with_top_line_corrected.pdf", format='pdf', bbox_inches='tight')
        plt.show()
